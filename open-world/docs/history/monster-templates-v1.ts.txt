// ==================== 怪物模板汇总 ====================

export const allMonsterTemplates: MonsterTemplate[] = [
  ...chapter1to3Monsters,
  ...chapter4Monsters,
  ...chapter5Monsters,
  ...chapter6Monsters
];

// 根据ID获取怪物模板
export function getMonsterTemplate(id: string): MonsterTemplate | undefined {
  return allMonsterTemplates.find(m => m.id === id);
}

// 根据等级计算怪物属性
export function calculateMonsterStats(template: MonsterTemplate, level: number): { hp: number; attack: number } {
  const hp = Math.floor(template.baseHp + template.hpPerLevel * (level - 1));
  const attack = Math.floor(template.baseAttack + template.attackPerLevel * (level - 1));
  return { hp, attack };
}

// 生成怪物spawn数据
export function createMonsterSpawn(
  template: MonsterTemplate,
  level: number,
  position: { x: number; y: number }
): Spawn {
  const stats = calculateMonsterStats(template, level);
  return {
    id: `${template.id}_${position.x}_${position.y}`,
    kind: template.behavior,
    name: template.name,
    x: position.x,
    y: position.y,
    hp: stats.hp,
    attack: stats.attack,
    elite: template.isElite || false,
    boss: template.isBoss || false,
    art: template.art
  };
}
